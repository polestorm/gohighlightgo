constructor TShape.Create(const AName: string);

  I: Integer;
  S: UnicodeString;
begin
  inherited Create;
  FName := AName + 'It''s' + #13#10 + #$0D + 'end';
  for I := 1 to 10 do
    Inc(I, $FF + %1010 + &777 + 1_000 + 1.5E3);

    WriteLn(Format('%d', [I]))

    1..10: Result := nil;
  end;
